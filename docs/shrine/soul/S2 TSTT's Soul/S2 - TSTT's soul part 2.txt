/**
 * ScreenManager — Build 57 P1 (Aetherweave 29) Quests Window Shell
 * DOM-agnostic core (node:test). Renderer only when `document` exists.
 * Modes: 'modal' (default) | 'full'.
 */
export class ScreenManager {
  constructor({ root = null } = {}) {
    this.root = root;
    this.screens = new Map();
    this.current = null;
    this.element = null;
  }

  register(id, { render } = {}) {
    this.screens.set(id, { render });
    return this;
  }

  open(id, mode = 'modal') {
    const screen = this.screens.get(id);
    if (!screen) return { ok: false };
    const safeMode = mode === 'full' ? 'full' : 'modal';
    this.current = { id, mode: safeMode, render: screen.render };
    this._render();
    return { ok: true, mode: safeMode };
  }

  setMode(mode) {
    if (!this.current) return { ok: false };
    if (mode !== 'modal' && mode !== 'full') return { ok: false };
    this.current.mode = mode;
    this._render();
    return { ok: true, mode };
  }

  close() {
    this.current = null;
    this._render();
    return { ok: true };
  }

  isOpen() { return Boolean(this.current); }
  getMode() { return this.current ? this.current.mode : null; }
  getScreenId() { return this.current ? this.current.id : null; }

  _render() {
    if (typeof document === 'undefined') return;
    if (this.element) { this.element.remove(); this.element = null; }
    if (!this.current) return;

    const { id, mode, render } = this.current;
    const body = typeof render === 'function' ? render() : '';

    const host = this.root || document.body;
    const wrapper = document.createElement('div');
    wrapper.className = mode === 'full' ? 'screen-full' : 'screen-modal';
    wrapper.dataset.screenId = id;

    const title = document.createElement('h2');
    title.textContent = mode === 'full' ? '📜 Quest Journal — Full' : '📜 Quest Journal';

    const bodyEl = document.createElement('div');
    bodyEl.className = 'screen-body';
    bodyEl.innerHTML = body;

    const actions = document.createElement('div');
    actions.className = 'screen-actions';

    const toggle = document.createElement('button');
    toggle.type = 'button';
    toggle.className = 'screen-toggle';
    toggle.textContent = mode === 'full' ? '⇱ Window' : '⛶ Full';
    toggle.addEventListener('click', () => this.setMode(mode === 'full' ? 'modal' : 'full'));

    const closeBtn = document.createElement('button');
    closeBtn.type = 'button';
    closeBtn.className = 'screen-close';
    closeBtn.textContent = 'Close';
    closeBtn.addEventListener('click', () => this.close());

    actions.append(toggle, closeBtn);
    wrapper.append(title, bodyEl, actions);
    host.append(wrapper);
    this.element = wrapper;
  }
}

import test from 'node:test';
import assert from 'node:assert/strict';
import { readFileSync } from 'node:fs';
import { dirname, join } from 'node:path';
import { fileURLToPath } from 'node:url';
import { SpriteAtlas } from '../src/rendering/SpriteAtlas.js';

const root = join(dirname(fileURLToPath(import.meta.url)), '..');
const weavers = JSON.parse(readFileSync(join(root, 'src/data/weavers.json'), 'utf8')).weavers;
const enemies = JSON.parse(readFileSync(join(root, 'src/data/enemies.json'), 'utf8')).enemies;

test('STYLE: every weaver id has a named placeholder color (not generic default)', () => {
  const atlas = new SpriteAtlas();
  for (const w of weavers) {
    const c = atlas.getColor(w.id);
    assert.notEqual(c, '#e4e4e4', `weaver ${w.id} missing placeholder — Forge must not ship unnamed ids`);
  }
});

test('STYLE: every enemy id has a named placeholder color (not generic default)', () => {
  const atlas = new SpriteAtlas();
  for (const e of enemies) {
    const c = atlas.getColor(e.id);
    assert.notEqual(c, '#e4e4e4', `enemy ${e.id} missing placeholder`);
  }
});

test('STYLE: unknown sprite id does not throw; getFrame is null', () => {
  const atlas = new SpriteAtlas();
  assert.doesNotThrow(() => atlas.getColor('gemini-hd2d-fake'));
  assert.equal(atlas.getFrame('gemini-hd2d-fake'), null);
  assert.equal(atlas.hasFrame('nope'), false);
});

test('STYLE: register records a frame; missing file stays unloadable without throwing getFrame', () => {
  const atlas = new SpriteAtlas();
  atlas.register('rudeus', '/sprites/does-not-exist.png');
  assert.equal(atlas.hasFrame('rudeus'), false);
  assert.equal(atlas.getFrame('rudeus'), null);
  const report = atlas.getLoadReport();
  assert.equal(report.registered, 1);
});

0. EXECUTIVE VERDICT

The Commander originally summarized the present condition as:

UI/UX needs a major upgrade, ChessLab needs a major reconfiguration, IdleHero needs a major update, and TAMAKEE needs major work, analysis, upgrades and better functions.

WISDOM agrees.

However, after cross-referencing the Commander’s observations against the current architecture, earlier Council work, and external systems, the problem is broader:

Marciale-OS has accumulated substantial functionality, but the maturity of its subsystem boundaries, lifecycle contracts, state observability, and cross-system integration has not kept pace with the growth of individual features.

Therefore:

VSS is not primarily a feature-development program. It is a system-cohesion and vertical-slice stabilization program.

The central question is no longer:

“Does this feature exist?”

It is:

“Can the Commander reliably use, observe, understand, recover, and trust the complete path from intent to result?”

1. EVIDENCE MODEL

Every finding in this audit must be classified as one of four evidence classes.

A — COMMANDER-OBSERVED

Direct observations reported by the Commander.

These are treated as legitimate defects requiring reproduction.

Current observations include:

IdleHero operates in the background, but its active session/lifesign is not visibly represented.

These observations must be reproduced and instrumented before implementation claims are made.

30. FINAL WISDOM THESIS

After combining both audits, I would put this sentence at the top of the VSS document:

Marciale-OS is no longer suffering primarily from a lack of functionality; it is suffering from uneven integration maturity.

The repository contains substantial functionality already. TAMAKEE likewise contains substantially more academic machinery than the visible interface currently exposes.

The next evolution should therefore not be:

Build more.

It should be:

Connect, stabilize, expose, verify, and recover what already exists.

The pattern repeats across nearly every VSS observation:

CAPABILITY EXISTS
       ↓
STATE EXISTS
       ↓
BOUNDARY / LIFECYCLE WEAK
       ↓
USER EXPERIENCE BREAKS
       ↓
OBSERVABILITY WEAK
       ↓
RECOVERY IS HARD

Therefore the central VSS doctrine should be:

CAPABILITY → CONTRACT → OBSERVABILITY → VERIFICATION → RECOVERY

A mature feature must satisfy all five.
